# External tools (not pip-installable) -- see README.md's Prerequisites
# section for the full install-and-run guide; summary here:
#
#   cutadapt >= 4.0        conda install -c bioconda cutadapt (or pip install cutadapt)
#   BLAST+ (blastn,        conda install -c bioconda blast
#     makeblastdb)
#   vsearch                conda install -c bioconda vsearch
#
# R (>= 4.3 recommended):
#   install.packages("BiocManager")
#   BiocManager::install("dada2")     # pulls in Biostrings, ShortRead, Rcpp, etc.
#                                      # first install compiles from source and
#                                      # can take 15-30 min.
#
# Reference database (SILVA v138.1, DADA2-formatted):
#   curl -LO https://zenodo.org/record/4587955/files/silva_nr99_v138.1_train_set.fa.gz
#   curl -LO https://zenodo.org/record/4587955/files/silva_species_assignment_v138.1.fa.gz
#   (~130 MB + ~75 MB; place under a silva/ directory and pass paths to run_dada2.R)
#   Several tools (resolve_unclassified_bacteria.py, rescue_unclassified_kingdom.py,
#   backfill_resolved_genus.py, check_split_chimeras.R) BLAST against SILVA directly --
#   build a BLAST database from the same file:
#     gunzip -k silva_nr99_v138.1_train_set.fa.gz -c > silva_nr99_v138.1.fasta
#     makeblastdb -in silva_nr99_v138.1.fasta -dbtype nucl -out blastdb/silva_nr99_v138.1
